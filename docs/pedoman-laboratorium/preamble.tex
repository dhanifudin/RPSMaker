    \draw[line width=1.2pt, color=polinemablue]
      ($(current page.north west)+(1.0cm,-1.25cm)$) --
      ($(current page.north east)+(-1.0cm,-1.25cm)$);
    \draw[line width=1.2pt, color=polinemablue]
      ($(current page.south west)+(1.0cm,1.1cm)$) --
      ($(current page.south east)+(-1.0cm,1.1cm)$);
  \end{tikzpicture}%
}
